**Step 1.1 — 定义索引数据结构（prefix_trie.py）**
```python
# 全局前缀树：block_hash 链 → 持有该前缀的实例集合
class GlobalPrefixTrie:
    def __init__(self):
        # block_hash -> {instance_id: last_seen_ts}
        self._hash_to_instances: dict[bytes, dict[str, float]] = {}

    def on_block_stored(self, instance_id: str, block_hashes: list[bytes], ts: float):
        for h in block_hashes:
            self._hash_to_instances.setdefault(h, {})[instance_id] = ts

    def on_block_removed(self, instance_id: str, block_hashes: list[bytes]):
        for h in block_hashes:
            self._hash_to_instances.get(h, {}).pop(instance_id, None)

    def match(self, block_hashes: list[bytes]) -> dict[str, int]:
        # 返回 {instance_id: 连续命中的前缀 block 数}
        counts: dict[str, int] = {}
        for h in block_hashes:  # 按前缀顺序，遇断即止
            holders = self._hash_to_instances.get(h)
            if not holders:
                break
            for inst in list(counts) or holders:
                if inst in holders:
                    counts[inst] = counts.get(inst, 0) + 1
        return counts
```

**Step 1.2 — 消费 kv_events（index_service.py）**
```python
# 订阅各实例的 ZmqEventPublisher 发出的 BlockStored/BlockRemoved
# 事件 schema 见 vllm/distributed/kv_events.py:48/92
import zmq, time
from prefix_trie import GlobalPrefixTrie

def run(sub_endpoints: list[str], query_port: int):
    trie = GlobalPrefixTrie()
    ctx = zmq.Context()
    sub = ctx.socket(zmq.SUB)
    for ep in sub_endpoints:  # 每个实例一个 endpoint
        sub.connect(ep)
    sub.setsockopt_string(zmq.SUBSCRIBE, "")
    # query 端口供 proxy 查询命中
    rep = ctx.socket(zmq.REP); rep.bind(f"tcp://*:{query_port}")
    poller = zmq.Poller(); poller.register(sub, zmq.POLLIN); poller.register(rep, zmq.POLLIN)
    while True:
        socks = dict(poller.poll())
        if sub in socks:
            inst_id, event = _parse_event(sub.recv_multipart())  # 解 BlockStored/Removed
            if event.type == "stored":
                trie.on_block_stored(inst_id, event.block_hashes, time.time())
            elif event.type == "removed":
                trie.on_block_removed(inst_id, event.block_hashes)
        if rep in socks:
            block_hashes = rep.recv_pyobj()
            rep.send_pyobj(trie.match(block_hashes))
```

**Step 3.2 — 联合决策（替换 _pick_server 的 prefill 选择）**
```python
def pick_prefiller_prefix_aware(self, request):
    block_hashes = _request_block_hashes(request.prompt_token_ids, self.block_size)
    hits = self._query_index(block_hashes)   # {instance: hit_blocks} via ZMQ REQ
    # 代价函数（对标 Dynamo router）：命中越多代价越低，负载越高代价越高
    def cost(inst):
        hit = hits.get(inst.key, 0)
        return self.alpha * (len(block_hashes) - hit) + self.beta * inst.active_tokens
    return min(self.prefill_servers, key=cost)   # 选代价最小
```

## 遗留问题（要求1）

1. **block hash 跨进程同源**：proxy 侧 `hash_request_tokens` 必须与 worker 侧产生相同 hash（含 cache_salt/LoRA/MM 维度，kv_events.py:64）。须实测验证一致性，否则查不中。❓
2. **索引 stale**：BlockRemoved 事件延迟 → 路由到已 evict 实例。需 TTL + connector load 失败兜底（get_block_ids_with_load_errors 已有）。🟡
3. **多 KV cache group**：`make_block_hash_with_group_id`(kv_cache_utils.py:56) 表明 hash 带 group id，索引需按 group 区分。❓
4. **512K 验证需多节点**：单节点 N=1 测不出跨实例增量，须 ≥2 实例。❓
